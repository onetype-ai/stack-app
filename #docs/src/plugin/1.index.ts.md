# index.ts

The public API: the only file another plugin may import, and only when it
names this one in `dependsOn`. Everything else in the folder is private.

`<Plugin>` is the plugin's folder capitalised.

```ts
import { usePlugin } from "@onetype/stack-app-kit/react";

import type { Context } from "@onetype/stack-app-kit";
import type { PluginHandle } from "@onetype/stack-app-kit/react";
import type { <Plugin> } from "./services/<plugin>";
import type { <Type> } from "./types/<Type>";

export type <Plugin>Services = { <service>: <Plugin> };

export type <Plugin>Handle = PluginHandle<<Plugin>Config, <Plugin>Services>;

const services = (ctx: Context) => ctx.use<<Plugin>Services>("<plugin>");

export const <Plugin> = {
    use: (): <Plugin>Handle => usePlugin<<Plugin>Config, <Plugin>Services>("<plugin>"),

    get: (ctx: Context, id: string): Promise<Type> =>
    {
        return services(ctx).<service>.get(id);
    },
};

export { <Plugin>Row } from "./components/<Plugin>Row/<Plugin>Row";
export type { <Type> } from "./types/<Type>";
```

`use()` is for a component, `get(ctx, ...)` for anything holding a `ctx`. A
component this plugin exports is shown the same way wherever it appears.
