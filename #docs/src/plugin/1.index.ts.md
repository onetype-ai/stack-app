# index.ts

The public API: the only file another plugin may import, and only when it
names this one in `dependsOn`. Everything else in the folder is private.

```ts
import { usePlugin } from "@onetype/stack-app-kit/react";

import type { Context } from "@onetype/stack-app-kit";
import type { ItemStore } from "./services/itemStore";
import type { Item } from "./types/Item";

export type ItemsServices = { items: ItemStore };

export const Items = {
    use: (): ItemStore => usePlugin<unknown, ItemsServices>("items").services.items,

    list: (ctx: Context): Promise<readonly Item[]> =>
    {
        return ctx.use<ItemsServices>("items").items.list();
    },
};

export { ItemRow } from "./components/ItemRow/ItemRow";
export type { Item } from "./types/Item";
```

`use()` is for a component, `list(ctx)` for anything holding a `ctx`. A
component exported here looks the same wherever it appears.
