# tests/

Through the page: what a reader sees for each answer a server can give, one
table a concern, never what the kit tests. A `usage.md` refusal is a row.

```tsx
import { screen } from "@testing-library/react";
import { expect, test } from "vitest";

import { ok, openApp } from "@onetype/stack-app-kit/testing/app";

import layout from "@plugins/layout/plugin";
import items from "../plugin";

const id = "5f0c7a3e-8b1d-4c2a-9e6f-1a2b3c4d5e6f";
const failed = "The items could not be loaded.";

test.each([
    { name: "lists each item", answer: ok({ items: [{ id, title: "First" }] }), shows: "First" },
    { name: "refuses a malformed list", answer: ok({ items: [{ id }] }), shows: failed },
    { name: "says the server failed", answer: { status: 503 }, shows: failed },
])("the page $name", async ({ answer, shows }) =>
{
    await openApp({ path: "/items", plugins: [layout, items], answers: { "GET /items": answer } });

    expect(await screen.findByText(shows)).toBeInTheDocument();
});
```

Break a row; watch it fail.
