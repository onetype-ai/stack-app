# tests/

Flat, one subject a file. `fakeContext` answers paths and records what was
asked, emitted and dropped.

```ts
import { fakeContext } from "@onetype/stack-app-kit/testing";
import { expect, test } from "vitest";

import { ItemStore } from "../services/itemStore";
import { ItemsKeys } from "../utils/ItemsKeys";

const item = { id: "5f0c7a3e-8b1d-4c2a-9e6f-1a2b3c4d5e6f", title: "First" };

test("create posts the title, drops the cached list and announces the id", async () =>
{
    const context = fakeContext({ "POST /items": item });

    await new ItemStore(context.ctx).create("First");

    expect(context.requests).toEqual([{ method: "POST", path: "/items", body: { title: "First" } }]);
    expect(context.invalidated).toEqual([ItemsKeys.list()]);
    expect(context.announced).toEqual([{ event: "items.created", payload: { id: item.id } }]);
});

test("list refuses a response that is not a list of items", async () =>
{
    const context = fakeContext({ "GET /items": { items: [{ id: item.id }] } });

    await expect(new ItemStore(context.ctx).list()).rejects.toThrow();
});
```

A render asserts through the role a reader uses, never a class name.

```tsx
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { expect, test, vi } from "vitest";

import { ItemRow } from "../components/ItemRow/ItemRow";

test("opening a row hands back that item's id", async () =>
{
    const onOpen = vi.fn();

    render(<ItemRow item={{ id: "9b1c", title: "First" }} onOpen={onOpen} />);

    await userEvent.click(screen.getByRole("button", { name: "First" }));

    expect(onOpen).toHaveBeenCalledWith("9b1c");
});
```

Break the code a test covers and watch it fail, then restore. A test that
never failed proves nothing.
