# tests/

Through the page: what a reader sees for each answer a server can give, one
table a concern, never what the kit tests. A `usage.md` refusal is a row.

```tsx
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { render, screen } from "@testing-library/react";
import { expect, test } from "vitest";

import { cache, createKernel } from "@onetype/stack-app-kit";
import { KernelProvider } from "@onetype/stack-app-kit/react";

import layout from "@plugins/layout/plugin";

import items from "../plugin";
import { ItemsPage } from "../pages/ItemsPage/ItemsPage";

const first = { id: "5f0c7a3e-8b1d-4c2a-9e6f-1a2b3c4d5e6f", title: "First" };

const failed = "The items could not be loaded.";

const cases = [
    { name: "lists each item", answer: () => Promise.resolve({ items: [first] }), shows: "First" },
    { name: "refuses a malformed list", answer: () => Promise.resolve({ items: [{ id: first.id }] }), shows: failed },
    { name: "says the server failed", answer: () => Promise.reject(new Error("503")), shows: failed },
];

test.each(cases)("the page $name", async ({ answer, shows }) =>
{
    const client = new QueryClient({ defaultOptions: { queries: { retry: false } } });
    const kernel = createKernel({
        plugins: [layout, items],
        http: { get: answer, post: answer, put: answer, patch: answer, delete: answer },
        cache: cache.fromQueries(client),
    });

    await kernel.start();
    render(
        <QueryClientProvider client={client}>
            <KernelProvider kernel={kernel}>
                <ItemsPage />
            </KernelProvider>
        </QueryClientProvider>,
    );

    expect(await screen.findByText(shows)).toBeInTheDocument();

    await kernel.stop();
});
```

Break a row; watch it fail.
