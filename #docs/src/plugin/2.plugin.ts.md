# plugin.ts

The contract: everything crossing the boundary, named here or absent. The
kernel refuses to start when a declaration is missing.

Required: `version`, `describe`. Every other key is optional.

```ts
export default definePlugin("<plugin>", {
    version: "1.0.0",
    describe: "<describe>",

    dependsOn: ["<other>"],
    config: <Plugin>Config.schema,

    permissions: { "<plugin>.read": { describe: "<describe>" } },

    // services before grants: inferred in order.
    services: (ctx) => ({ <service>: new <Plugin>(ctx) }),
    grants: (ctx) => ctx.services.<service>.permissions(),

    frame: <Plugin>Frame,
    pages: { forbidden: NoEntry, missing: NoPage },

    routes: [
        { path: "/<plugin>", title: "<describe>", requires: ["<plugin>.read"],
          component: <Plugin>s, search: <Plugin>Query.schema },
        { path: "/<plugin>/$id", title: "<describe>", component: <Plugin>Detail },
    ],

    slots: { "<plugin>.<place>": { describe, schema } },
    contributes: [{ slot: "<other>.<place>", render: <Plugin>Link }],

    emits: { "<plugin>.<event>": { describe, schema } },
    listens: { "<other>.<event>": { describe, handle } },
    hooks: { "<plugin>.before-<action>": { describe, schema } },
    participates: { "<other>.before-<action>": { describe, handle } },
    commands: { "<plugin>.<action>": { describe, schema, requires, run } },

    sends: (ctx) => ({ "x-<plugin>": ctx.config.<field> }),
    fallback: <Plugin>Broke,

    setup: (ctx) =>
    {
        ctx.log.info("<plugin> ready");
    },
    teardown: (ctx) =>
    {
        ctx.log.info("<plugin> stopped");
    },
});
```

`frame` wraps every page this plugin routes, `pages` replaces their 403 and
404, and `fallback` renders when one throws.
