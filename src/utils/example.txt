==> src/utils/Words.ts

class WordUtils
{
    count(words: number, locale?: string): string
    {
        if (!Number.isFinite(words))
        {
            return "";
        }

        const rule = new Intl.PluralRules(locale);
        const name = rule.select(words) === "one" ? "word" : "words";

        return `${new Intl.NumberFormat(locale).format(words)} ${name}`;
    }

    minutes(words: number, perMinute = 200): number
    {
        if (!Number.isFinite(words) || words <= 0)
        {
            return 0;
        }

        return Math.max(1, Math.ceil(words / perMinute));
    }
}

export const Words = new WordUtils();
