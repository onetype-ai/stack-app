==> src/utils/Money.ts

class MoneyUtils
{
    /* Cents divided at the edge, never a float carried through the app: 0.1
       plus 0.2 must never reach a total somebody is asked to pay. */
    format(cents: number, currency: string, locale?: string): string
    {
        if (!Number.isFinite(cents))
        {
            return "";
        }

        return new Intl.NumberFormat(locale, { style: "currency", currency }).format(cents / 100);
    }
}

export const Money = new MoneyUtils();
