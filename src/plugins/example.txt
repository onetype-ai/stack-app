import { definePlugin } from "@onetype/stack-app-kit";

import { DocumentDetail } from "./pages/DocumentDetail/DocumentDetail";
import { Documents } from "./pages/Documents/Documents";
import { DeskFrame } from "./sections/DeskFrame/DeskFrame";
import { NoEntry } from "./sections/NoEntry/NoEntry";
import { NoPage } from "./sections/NoPage/NoPage";
import { createDocumentsService } from "./services/documents";
import { createReaderService } from "./services/reader";
import { Beside } from "./types/Beside";
import { DocumentQuery } from "./types/DocumentQuery";
import { DocumentsConfig } from "./types/DocumentsConfig";
import { Filing } from "./types/Filing";
import { NavPlace } from "./types/NavPlace";

export default definePlugin("documents", {
    version: "1.0.0",

    describe: "What is on the desk: documents, how long each runs, and what is open on them.",

    dependsOn: [],

    config: DocumentsConfig.schema,

    permissions: {
        "documents.read": { describe: "See what is on the desk." },
        "documents.file": { describe: "File a document away." },
    },

    services: (ctx) =>
    {
        return {
            documents: createDocumentsService(ctx),
            reader: createReaderService(ctx),
        };
    },

    grants: (ctx) =>
    {
        return ctx.services.reader.permissions();
    },

    frame: DeskFrame,

    pages: {
        forbidden: NoEntry,
        missing: NoPage,
    },

    routes: [
        {
            path: "/documents",
            title: "Desk",
            requires: ["documents.read"],
            component: Documents,
            search: DocumentQuery.schema,
        },
        {
            path: "/documents/$id",
            title: "Document",
            requires: ["documents.read"],
            component: DocumentDetail,
        },
    ],

    slots: {
        "documents.nav": {
            describe: "A way into another plugin, in the bar. Contribute a link; the desk only places it.",
            schema: NavPlace.schema,
        },

        "documents.beside": {
            describe: "What another plugin shows beside one document, given its id and title.",
            schema: Beside.schema,
        },
    },

    emits: {
        "documents.filed": {
            describe: "A document was filed away. Emitted after the filing was written, never before.",
            schema: Filing.schema,
        },
    },

    hooks: {
        "documents.before-file": {
            describe: "Runs before a document is filed. Returning a string refuses it with that reason.",
            schema: Filing.schema,
        },
    },

    setup: async (ctx) =>
    {
        await ctx.services.reader.load();

        ctx.log.info("documents ready", {
            locale: ctx.config.locale,
            atTheDesk: ctx.services.reader.current()?.displayName ?? "nobody",
        });
    },

    teardown: (ctx) =>
    {
        ctx.log.info("documents stopped");
    },
});

==> src/plugins/documents/usage.md

# documents

## Description

What is on the desk: the documents, how long each runs, and filing one away.

## Purpose

The documents are this plugin's and nobody else's. Another plugin that needs a
title asks for it rather than holding its own copy, so one name cannot go
stale in two places.

Filing is a hook rather than a command anyone may run: whoever holds a
document has a say before it leaves the desk, and says why.

## Usage

```ts
import { Documents } from "@plugins/documents";

const title = await Documents.titleOf(ctx, id);
const document = await Documents.get.one(ctx, id);
const page = await Documents.get.page(ctx, { kind: "report" });
```

In a component, `Documents.use()` answers the plugin's handle. `DocumentRow`
is exported so a document looks the same wherever it is shown.

## What it declares

- **Permissions**: `documents.read` to see the desk, `documents.file` to file.
- **Routes**: `/documents` lists them, `/documents/$id` opens one.
- **Slot** `documents.beside`: what another plugin shows beside a document.
- **Event** `documents.filed`, after the filing is written.
- **Hook** `documents.before-file`: return a string to refuse it.

## Refuses

- Reaching a document through anything but the public entry.
- Filing without running the hook first.

import { Document } from "../types/Document";
import { DocumentPage } from "../types/DocumentPage";
import type { DocumentQuery } from "../types/DocumentQuery";

export const documentsApi = {
    list: async (ctx: Context, query: DocumentQuery): Promise<DocumentPage> =>
    {
        const answered = await ctx.http.get("/documents", {
            query: { ...(query.kind !== undefined && { kind: query.kind }) },
        });

        return DocumentPage.schema.parse(answered);
    },

    one: async (ctx: Context, id: string): Promise<Document> =>
    {
        const answered = await ctx.http.get(`/documents/${encodeURIComponent(id)}`);

        return Document.schema.parse(answered);
    },

    file: async (ctx: Context, id: string): Promise<Document> =>
    {
        const answered = await ctx.http.post(`/documents/${encodeURIComponent(id)}/filing`);

        return Document.schema.parse(answered);
    },
};

import { DocumentKind } from "../../types/DocumentKind";
import { Words } from "@utils/Words";
import styles from "./DocumentRow.module.css";

import type { ReactNode } from "react";
import type { Document } from "../../types/Document";

export type DocumentRowProps = {
    document: Document;
    locale: string;
    longRead: number;
    href?: string | undefined;
    trailing?: ReactNode;
    onOpen?: ((id: string) => void) | undefined;
};

export const DocumentRow = ({ document, locale, longRead, href, trailing, onOpen }: DocumentRowProps) =>
{
    const length = document.words === 0 ? "empty" : document.words >= longRead ? "long" : "short";

    return (
        <li className={styles.root} data-length={length}>
            <div className={styles.identity}>
                {href === undefined
                    ? <span className={styles.name}>{document.title}</span>
                    : (
                        <a
                            className={styles.name}
                            href={href}
                            onClick={(event) =>
                            {
                                if (onOpen === undefined || event.metaKey || event.ctrlKey)
                                {
                                    return;
                                }

                                event.preventDefault();
                                onOpen(document.id);
                            }}
                        >
                            {document.title}
                        </a>
                    )}

                <span className={styles.kind}>{DocumentKind.label(document.kind)}</span>
            </div>

            <span className={styles.length}>{Words.count(document.words, locale)}</span>

            <span className={styles.open} data-open={document.openComments > 0 || undefined}>
                {document.openComments === 0
                    ? "No comments"
                    : `${String(document.openComments)} open`}
            </span>

            <div className={styles.trailing}>{trailing}</div>
        </li>
    );
};

import { useNavigate, useSearch } from "@tanstack/react-router";

import { DocumentQuery } from "../types/DocumentQuery";

import type { DocumentKind } from "../types/DocumentKind";

export type KindInAddress = {
    kind: DocumentKind | undefined;
    choose: (kind: DocumentKind | undefined) => void;
};

export const useKindInAddress = (): KindInAddress =>
{
    const navigate = useNavigate();
    const search: unknown = useSearch({ strict: false });
    const query = DocumentQuery.schema.parse(search);

    return {
        ...(query.kind !== undefined ? { kind: query.kind } : { kind: undefined }),

        choose: (kind: DocumentKind | undefined): void =>
        {
            void navigate({ to: "/documents", search: kind === undefined ? {} : { kind }, replace: true });
        },
    };
};

import { useMutation, useQuery } from "@tanstack/react-query";
import { useNavigate, useParams } from "@tanstack/react-router";

import { Slot } from "@onetype/stack-app-kit/react";

import { Documents } from "../../index";
import { DocumentId } from "../../types/DocumentId";
import { DocumentKind } from "../../types/DocumentKind";
import { DocumentKeys } from "../../utils/DocumentKeys";
import { Words } from "@utils/Words";
import styles from "./DocumentDetail.module.css";

export const DocumentDetail = () =>
{
    const params: unknown = useParams({ strict: false });
    const id = DocumentId.parse((params as Record<string, unknown>)["id"]);
    const { services, config, permissions } = Documents.use();
    const navigate = useNavigate();

    const document = useQuery({
        queryKey: DocumentKeys.one(id),
        queryFn: () => services.documents.one(id),
    });

    const file = useMutation({
        mutationFn: () => services.documents.file(id),
        onSuccess: () => { void navigate({ to: "/documents" }); },
    });

    return (
        <article className={styles.root}>
            <header>
                <p className={styles.eyebrow}>
                    {document.data === undefined ? "Document" : DocumentKind.label(document.data.kind)}
                </p>

                <h1 className={styles.title}>
                    {document.isPending ? <span className={styles.bar} aria-hidden="true" /> : document.data?.title ?? "Unknown document"}
                </h1>
            </header>

            {document.isError && (
                <p className={styles.wrong} role="alert">
                    This document could not be read. {document.error.message}
                </p>
            )}

            {document.isSuccess && (
                <div className={styles.body}>
                    <dl className={styles.facts}>
                        <div className={styles.fact}>
                            <dt className={styles.term}>Length</dt>
                            <dd className={styles.value}>
                                {Words.count(document.data.words, config.locale)}
                                {", about "}
                                {String(Words.minutes(document.data.words))} min
                            </dd>
                        </div>

                        <div className={styles.fact}>
                            <dt className={styles.term}>Still open</dt>
                            <dd className={styles.value}>
                                <span className={styles.open} data-open={document.data.openComments > 0 || undefined}>{document.data.openComments === 0
                                        ? "Nothing open"
                                        : `${String(document.data.openComments)} comments`}</span>
                            </dd>
                        </div>
                    </dl>

                    <aside className={styles.aside} aria-label="What else this document is part of">
                        <Slot name="documents.beside" payload={{ id: document.data.id, title: document.data.title }} />
                    </aside>

                    {permissions.has("documents.file") && (
                        <footer className={styles.foot}>
                            <button type="button" className={styles.action} aria-busy={file.isPending || undefined} disabled={file.isPending}
                                onClick={() => { file.mutate(); }}
                            >
                                File it away
                            </button>

                            {file.isError && (
                                <p className={styles.refused} role="alert">{file.error.message}</p>
                            )}
                        </footer>
                    )}
                </div>
            )}
        </article>
    );
};

import type { Context } from "@onetype/stack-app-kit";

import { documentsApi } from "../api/documents";
import type { Document } from "../types/Document";
import type { DocumentPage } from "../types/DocumentPage";
import { DocumentQuery } from "../types/DocumentQuery";
import { Filing } from "../types/Filing";
import { DocumentKeys } from "../utils/DocumentKeys";

export const createDocumentsService = (ctx: Context) =>
{
    const known = new Map<string, Document>();

    const remember = (document: Document): Document =>
    {
        known.set(document.id, document);

        return document;
    };

    return {
        cached: (id: string): Document | undefined =>
        {
            return known.get(id);
        },

        list: async (query: Partial<DocumentQuery> = {}): Promise<DocumentPage> =>
        {
            const page = await documentsApi.list(ctx, DocumentQuery.schema.parse(query));

            for (const document of page.documents)
            {
                remember(document);
            }

            return page;
        },

        one: async (id: string): Promise<Document> =>
        {
            return remember(await documentsApi.one(ctx, id));
        },

        titleOf: async (id: string): Promise<string> =>
        {
            const held = known.get(id);

            if (held !== undefined)
            {
                return held.title;
            }

            return remember(await documentsApi.one(ctx, id)).title;
        },

        file: async (id: string): Promise<void> =>
        {
            const held = known.get(id);
            const document = held ?? remember(await documentsApi.one(ctx, id));
            const filing = Filing.schema.parse({ id: document.id, title: document.title });
            const refused = await ctx.hooks.run("documents.before-file", filing);

            if (refused !== undefined)
            {
                throw new Error(`This document cannot be filed: ${refused}`);
            }

            await documentsApi.file(ctx, id);

            known.delete(id);

            ctx.cache.invalidate(DocumentKeys.all());
            ctx.events.emit("documents.filed", filing);
        },
    };
};

export type DocumentsService = ReturnType<typeof createDocumentsService>;

                const gone = Filed.schema.parse(announced);

                ctx.services.drafting.strike(gone.id);
            },
        },
    },

    participates: {
        "documents.before-file": {
            describe: "Refuses to file a document somebody still has a draft against.",
            handle: (asked, ctx) =>
            {
                const document = Filing.schema.parse(asked);

                return ctx.services.drafting.holds(document.id)
                    ? `"${document.title}" carries an unsent draft, so it cannot be filed yet.`

            },
        },

            {

                const list = ctx.services.drafting.read();

                {

        },
    },

    {
        ctx.log.info("comments ready", { maxDrafts: ctx.config.maxDrafts });
    },

    teardown: (ctx) =>
    {
        ctx.log.info("comments stopped", { left: ctx.services.drafting.read().items });
    },
});

==> src/plugins/comments/usage.md

# comments

## Description

Comments somebody means to make, held on the desk until a board reads them.

## Purpose

A draft is kept here, not on the server: it belongs to whoever is typing it
and to nobody else until it is sent. So the drafts live in a service, and a
view reads them through `useStore` rather than through a query.

The plugin never learns what a document is. It asks `documents` for a title
and renders the desk's own row, so a document looks the same on both screens.

## Usage

```ts
import { Comments } from "@plugins/comments";

const drafts = Comments.drafts.all(ctx);

await Comments.drafts.add(ctx, documentId, "Needs a date on it.");
await Comments.publish(ctx, "review");
```

In a component, `Comments.use()` answers the plugin's handle.

## What it declares

- **Permission**: `comments.write` to draft and to send.
- **Routes**: `/comments` lists the drafts, `/comments/publish` sends them.
  The second answers `instead` when nothing is drafted, so an empty desk never
  opens a form with nothing to submit.
- **Fills** `documents.beside` with a box that drafts a comment.
- **Hears** `documents.filed` and strikes the draft against it.
- **Joins** `documents.before-file` and refuses while a draft is unsent.
- **Command** `comments.publish`: sends every draft and empties the desk.

## Refuses

- Two drafts against one document.
- Sending when nothing is drafted.

import { useState } from "react";

import { Board } from "../types/Board";

export type PublishForm = {
    board: string;
    boardName: string;
    ready: boolean;
    sending: boolean;
    problem: string | undefined;
    type: (typed: string) => void;
    ask: () => void;
    drop: () => void;
    refuse: (cause: unknown) => void;
};

export const usePublishForm = (): PublishForm =>
{
    const [board, setBoard] = useState("");
    const [sending, setSending] = useState(false);
    const [problem, setProblem] = useState<string | undefined>(undefined);

    const boardName = board.trim().toLowerCase();
    const ready = Board.schema.safeParse(boardName).success;

    return {
        board,
        boardName,
        ready,
        sending,
        problem,

        type: (typed: string): void =>
        {
            setProblem(undefined);
            setBoard(typed);
        },

        ask: (): void =>
        {
            setSending(true);
        },

        drop: (): void =>
        {
            setSending(false);
        },

        refuse: (cause: unknown): void =>
        {
            setSending(false);
            setProblem(cause instanceof Error ? cause.message : String(cause));
        },
    };
};

import { useState } from "react";

import { useEvent, useStore } from "@onetype/stack-app-kit/react";

import { Comments } from "../../index";
import styles from "./AddComment.module.css";

import type { Beside, Filing } from "@plugins/documents";

export type AddCommentProps = {
    payload: unknown;
};

export const AddComment = ({ payload }: AddCommentProps) =>
{
    const { services } = Comments.use();
    const document = payload as Beside;

    const [body, setBody] = useState("");
    const [problem, setProblem] = useState<string | undefined>(undefined);
    const [filed, setFiled] = useState(false);

    const drafts = useStore(services.drafting.watch, services.drafting.read);
    const already = drafts.drafts.find((draft) => draft.documentId === document.id && !draft.gone);

    useEvent("comments", "documents.filed", (announced) =>
    {
        if ((announced as Filing).id === document.id)
        {
            setFiled(true);
        }
    });

    const add = (): void =>
    {
        setProblem(undefined);

        services.drafting.add(document.id, body)
            .then(() => { setBody(""); })
            .catch((cause: unknown) =>
            {
                setProblem(cause instanceof Error ? cause.message : String(cause));
            });
    };

    return (
        <div className={styles.root}>
            <div className={styles.line}>
                <p className={styles.label}>Comment</p>

                {already !== undefined && <span className={styles.drafted}>Drafted</span>}
            </div>

            {already === undefined && (
                <label className={styles.field}>
                    <span className={styles.what}>What to say</span>

                    <textarea
                        className={styles.body}
                        value={body}
                        rows={3}
                        disabled={filed}
                        onChange={(event) => { setBody(event.target.value); }}
                    />
                </label>
            )}

            <button type="button" className={styles.action} disabled={filed || already !== undefined} onClick={add}>
                {already === undefined ? "Draft a comment" : "Already drafted"}
            </button>

            {filed && (
                <p className={styles.note} role="status">
                    This document was just filed away, so nothing more can be said on it.
                </p>
            )}

            {problem !== undefined && <p className={styles.problem} role="alert">{problem}</p>}
        </div>
    );
};

import { Documents } from "@plugins/documents";

import type { Context } from "@onetype/stack-app-kit";
import type { CommentsConfig } from "../types/CommentsConfig";
import type { Draft } from "../types/Draft";
import type { DraftList } from "../types/DraftList";
import { Drafts } from "../utils/Drafts";

export const createDraftingService = (ctx: Context<CommentsConfig>) =>
{
    let drafts: readonly Draft[] = [];

    const watchers = new Set<() => void>();

    let snapshot: DraftList = { drafts: [], words: 0, items: 0 };

    const publish = (next: readonly Draft[]): void =>
    {
        drafts = next;
        snapshot = { drafts: [...next], words: Drafts.words(next), items: Drafts.items(next) };

        for (const watcher of watchers)
        {
            watcher();
        }
    };

    return {
        read: (): DraftList =>
        {
            return snapshot;
        },

        watch: (notify: () => void): (() => void) =>
        {
            watchers.add(notify);

            return () =>
            {
                watchers.delete(notify);
            };
        },

        holds: (documentId: string): boolean =>
        {
            return Drafts.holds(drafts, documentId);
        },

        add: async (documentId: string, body: string): Promise<void> =>
        {
            if (body.trim() === "")
            {
                throw new Error("A comment with nothing in it says nothing.");
            }

            if (Drafts.count(body) > ctx.config.maxLength)
            {
                throw new Error(`A comment may not run past ${String(ctx.config.maxLength)} words.`);
            }

            if (Drafts.holds(drafts, documentId))
            {
                throw new Error("This document already carries a draft of yours.");
            }

            if (drafts.length >= ctx.config.maxDrafts)
            {
                throw new Error(`A desk may not hold more than ${String(ctx.config.maxDrafts)} drafts.`);
            }

            const title = await Documents.titleOf(ctx, documentId);

            const now = drafts;

            if (Drafts.holds(now, documentId))
            {
                return;
            }

            if (now.length >= ctx.config.maxDrafts)
            {
                throw new Error(`A desk may not hold more than ${String(ctx.config.maxDrafts)} drafts.`);
            }

            publish([...now, { documentId, title, body, gone: false }]);
        },

        drop: (documentId: string): void =>
        {
            publish(drafts.filter((draft) => draft.documentId !== documentId));
        },

        strike: (documentId: string): void =>
        {
            if (!Drafts.holds(drafts, documentId))
            {
                return;
            }

            publish(drafts.map((draft) =>
            {
                return draft.documentId === documentId ? { ...draft, gone: true } : draft;
            }));
        },

        empty: (): void =>
        {
            publish([]);
        },
    };
};

export type DraftingService = ReturnType<typeof createDraftingService>;

import type { Draft } from "../types/Draft";

class DraftUtils
{
    words(drafts: readonly Draft[]): number
    {
        return drafts.reduce((sum, draft) => sum + this.count(draft.body), 0);
    }

    items(drafts: readonly Draft[]): number
    {
        return drafts.length;
    }

    holds(drafts: readonly Draft[], documentId: string): boolean
    {
        return drafts.some((draft) => draft.documentId === documentId);
    }

    count(body: string): number
    {
        return body.trim() === "" ? 0 : body.trim().split(/\s+/).length;
    }
}

export const Drafts = new DraftUtils();
