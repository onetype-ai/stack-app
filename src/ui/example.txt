# ui

Shared presentational layer. Knows no domain, imports no plugin.

```ts
import { Example } from "@ui";
import "@ui/styles/index.css";
```

## What is here, and what is not

One unit, `Example`, and eighteen tokens named `--example1` upward. Neither is
a decision: the names say nothing about a role and the values say nothing at
all, so nothing here can be copied into a real application by accident.

The whole folder folds into `example.txt`. Read it, take the mechanics, and
write your own: the palette, the type scale and the units your application
needs are yours to choose on the first day.

## Layout

- `components/`: smallest units. Stateless, no domain knowledge.
- `sections/`: blocks composed of components. Hold their own UI state, never
  load data.
- `styles/`: `reset` clears, `tokens` declares, `base` applies. Imported once,
  in that order, through `styles/index.css`.
- `index.ts`: the only entry. A plugin imports `@ui`, never a path inside it.

## Rules

Styling comes from tokens: a literal colour, length or duration in a module is
a token that was not declared. `Project.checks()` refuses both a `var()`
nothing declares and a `styles.name` no module carries.

Props in, markup out. A unit never fetches, never reads global state, and never
knows what sits above it. Keyboard-operable, visible focus, an empty case.

## Adding a unit

1. Create `Name/` holding `Name.tsx` and `Name.module.css`.
2. Export it from `index.ts`.
3. Test it through its public surface, then break the behaviour and watch the
   test fail. One that stays green proves nothing.

For what a unit accepts, read its exported prop type. The type is the contract.

==> src/ui/components/Example/Example.tsx

import { useId, useRef } from "react";

import { useDismiss, useFocusTrap } from "@onetype/stack-app-kit/react";

import styles from "./Example.module.css";

import type { ReactNode } from "react";

export type ExampleTone = "plain" | "accent" | "alarm";

export type ExampleProps = {
    title: string;
    tone?: ExampleTone;
    note?: string;
    problem?: string;
    busy?: boolean;
    dismissible?: boolean;
    onDismiss?: () => void;
    children?: ReactNode;
    footer?: ReactNode;
};

export const Example = ({
    title,
    tone = "plain",
    note,
    problem,
    busy = false,
    dismissible = false,
    onDismiss,
    children,
    footer,
}: ExampleProps) =>
{
    const id = useId();
    const region = useRef<HTMLElement>(null);

    useFocusTrap(dismissible, region);
    useDismiss(dismissible, region, undefined, onDismiss ?? (() => undefined));

    const describedBy = [note !== undefined ? `${id}-note` : "", problem !== undefined ? `${id}-problem` : ""]
        .filter(Boolean)
        .join(" ");

    return (
        <section
            ref={region}
            className={styles.root}
            data-tone={tone}
            data-busy={busy || undefined}
            aria-labelledby={id}
            aria-describedby={describedBy === "" ? undefined : describedBy}
            aria-busy={busy || undefined}
        >
            <header className={styles.head}>
                <h2 id={id} className={styles.title}>{title}</h2>

                {dismissible && (
                    <button
                        type="button"
                        className={styles.dismiss}
                        onClick={onDismiss}
                        aria-label={`Dismiss ${title}`}
                    >
                        ✕
                    </button>
                )}
            </header>

            {note !== undefined && <p id={`${id}-note`} className={styles.note}>{note}</p>}

            {busy
                ? (
                    <div className={styles.waiting} aria-hidden="true">
                        <span className={styles.bar} />
                        <span className={styles.bar} />
                        <span className={styles.bar} />
                    </div>
                )
                : <div className={styles.body}>{children}</div>}

            {problem !== undefined && (
                <p id={`${id}-problem`} className={styles.problem} role="alert">{problem}</p>
            )}

            {footer !== undefined && <footer className={styles.foot}>{footer}</footer>}
        </section>
    );
};

==> src/ui/tests/units.test.tsx

import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { describe, expect, test, vi } from "vitest";

import { Example } from "../index";

describe("a region a reader is meant to act on", () =>
{
    test("names itself by its own heading, so a screen reader can jump to it", () =>
    {
        render(<Example title="What is waiting" />);

        expect(screen.getByRole("region", { name: "What is waiting" })).toBeDefined();
    });

    test("ties its note to the region, so it is read with the heading and not after it", () =>
    {
        render(<Example title="What is waiting" note="Nothing has been sent yet." />);

        const said = screen.getByRole("region", { name: "What is waiting" }).getAttribute("aria-describedby") ?? "";

        expect(said).not.toBe("");
        expect(document.getElementById(said.split(" ")[0] ?? "")?.textContent).toBe("Nothing has been sent yet.");
    });
});

describe("a region that is still loading", () =>
{
    test("says it is busy, so nothing reads as an empty one", () =>
    {
        render(<Example title="What is waiting" busy />);

        expect(screen.getByRole("region", { name: "What is waiting" }).getAttribute("aria-busy")).toBe("true");
    });

    test("and hides what stands in for the content, so a reader hears it once", () =>
    {
        const { container } = render(<Example title="What is waiting" busy>Real content</Example>);

        expect(screen.queryByText("Real content")).toBeNull();
        expect(container.querySelector("[aria-hidden='true']")).not.toBeNull();
    });
});

describe("a region with something wrong", () =>
{
    test("announces the problem rather than only colouring a border", () =>
    {
        render(<Example title="What is waiting" problem="That board does not exist." />);

        expect(screen.getByRole("alert").textContent).toBe("That board does not exist.");
    });

    test("and ties it to the region as well as the note", () =>
    {
        render(<Example title="What is waiting" note="Pick a board." problem="That board does not exist." />);

        const said = screen.getByRole("region", { name: "What is waiting" }).getAttribute("aria-describedby") ?? "";

        expect(said.split(" ")).toHaveLength(2);
    });
});

describe("a region a reader may dismiss", () =>
{
    test("offers a control that names what it closes", () =>
    {
        render(<Example title="What is waiting" dismissible onDismiss={() => undefined} />);

        expect(screen.getByRole("button", { name: "Dismiss What is waiting" })).toBeDefined();
    });

    test("and says so when it is pressed", async () =>
    {
        const dismissed = vi.fn();

        render(<Example title="What is waiting" dismissible onDismiss={dismissed} />);

        await userEvent.click(screen.getByRole("button", { name: "Dismiss What is waiting" }));

        expect(dismissed).toHaveBeenCalledOnce();
    });

    test("while one nobody may dismiss offers no control at all", () =>
    {
        render(<Example title="What is waiting" />);

        expect(screen.queryByRole("button")).toBeNull();
    });
});
